# Copycat Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a Windows desktop clipboard history manager (Electron +
React + TypeScript) with categorized capture (text/image/file), search,
tags, global hotkey, and a tray presence, matching the `raw/template.png`
reference UI.

**Architecture:** Electron main process owns clipboard polling, SQLite
storage, and OS integration (tray, global hotkey, window). A typed
`contextBridge` preload exposes an IPC API. The React renderer renders
history as a virtualized horizontal card grid with search, tag filters,
and a Zustand store that talks to the preload API.

**Tech Stack:** Electron, electron-vite, React 18, TypeScript,
better-sqlite3, Zustand, react-window, Jest + ts-jest + Testing Library,
Playwright (e2e), electron-builder (Windows NSIS packaging).

**Spec:** `docs/superpowers/specs/2026-10-13-copycat-design.md`

## Global Constraints

- Windows-only target; OS integration code (tray, global hotkey,
  `FileNameW` clipboard format) assumes Windows.
- SQLite DB at `%APPDATA%/Copycat/db.sqlite`; image previews at
  `%APPDATA%/Copycat/previews/`.
- Files are referenced by path, never copied into app storage.
- Default retention: 1000 non-starred items (`DEFAULT_RETENTION_LIMIT`
  in `src/shared/config.ts`); starred items are never auto-pruned.
- Default global hotkey: `CommandOrControl+Shift+V`
  (`DEFAULT_HOTKEY` in `src/shared/config.ts`).
- Max stored text length: 200,000 characters
  (`MAX_CONTENT_LENGTH` in `src/shared/config.ts`); longer text is
  truncated at insert time.
- Out of scope (do not build): cloud sync, in-place content editing,
  history encryption, sensitive-data (password) detection.

## Review Focus

- Empty or whitespace-only text copies must not create history items —
  covered in Task 5 (`ClipboardWatcher` skips blank text).
- Very large copied text must not grow the DB unbounded — covered in
  Task 3 (`insertItem` truncates to `MAX_CONTENT_LENGTH`).
- Malformed/garbage `FileNameW` clipboard buffers must not crash
  capture — covered in Task 4 (`parseFileDropBuffer` fails closed to
  `[]`).
- Tags differing only by case/whitespace must not create duplicate tag
  rows ("Api" vs "api ") — covered in Task 3 (`tags-repo` normalizes
  tag names).
- Global hotkey registration failing (already bound by another app)
  must not crash the app or leave it with no way to open — covered in
  Task 8 (`registerHotkey` returns a success boolean and logs instead
  of throwing; tray `Open` menu item always works).

```ts
import Database from 'better-sqlite3';
import fs from 'node:fs';
import path from 'node:path';

const SCHEMA = `
CREATE TABLE IF NOT EXISTS items (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  type TEXT NOT NULL CHECK(type IN ('text','image','file')),
  content TEXT NOT NULL,
  source_app TEXT,
  created_at INTEGER NOT NULL,
  starred INTEGER NOT NULL DEFAULT 0,
  pinned INTEGER NOT NULL DEFAULT 0
);

CREATE TABLE IF NOT EXISTS tags (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  name TEXT NOT NULL UNIQUE
);

CREATE TABLE IF NOT EXISTS item_tags (
  item_id INTEGER NOT NULL REFERENCES items(id) ON DELETE CASCADE,
  tag_id INTEGER NOT NULL REFERENCES tags(id) ON DELETE CASCADE,
  PRIMARY KEY (item_id, tag_id)
);

CREATE INDEX IF NOT EXISTS idx_items_created_at ON items(created_at DESC);
CREATE INDEX IF NOT EXISTS idx_items_starred ON items(starred);
`;

export function createDb(dbPath: string): Database.Database {
  if (dbPath !== ':memory:') {
    fs.mkdirSync(path.dirname(dbPath), { recursive: true });
  }
  const db = new Database(dbPath);
  db.pragma('foreign_keys = ON');
  db.exec(SCHEMA);
  return db;
}
```

```ts
import type Database from 'better-sqlite3';

export interface TagCount {
  name: string;
  count: number;
}

export interface TagsRepo {
  addTagToItem(itemId: number, tagName: string): void;
  removeTagFromItem(itemId: number, tagName: string): void;
  getTagsForItem(itemId: number): string[];
  getAllTagsWithCounts(): TagCount[];
}

function normalizeTagName(name: string): string {
  return name.trim().toLowerCase();
}

export function createTagsRepo(db: Database.Database): TagsRepo {
  return {
    addTagToItem(itemId, tagName) {
      const name = normalizeTagName(tagName);
      if (!name) return;
      db.prepare('INSERT OR IGNORE INTO tags (name) VALUES (?)').run(name);
      const tag = db.prepare('SELECT id FROM tags WHERE name = ?').get(name) as { id: number };
      db.prepare('INSERT OR IGNORE INTO item_tags (item_id, tag_id) VALUES (?, ?)').run(
        itemId,
        tag.id,
      );
    },
    removeTagFromItem(itemId, tagName) {
      const name = normalizeTagName(tagName);
      db.prepare(
        `DELETE FROM item_tags WHERE item_id = ? AND tag_id = (SELECT id FROM tags WHERE name = ?)`,
      ).run(itemId, name);
    },
    getTagsForItem(itemId) {
      return (
        db
          .prepare(
            `SELECT tags.name as name FROM tags
             JOIN item_tags ON item_tags.tag_id = tags.id
             WHERE item_tags.item_id = ?
             ORDER BY tags.name`,
          )
          .all(itemId) as { name: string }[]
      ).map((r) => r.name);
    },
    getAllTagsWithCounts() {
      return db
        .prepare(
          `SELECT tags.name as name, COUNT(item_tags.item_id) as count
           FROM tags LEFT JOIN item_tags ON item_tags.tag_id = tags.id
           GROUP BY tags.id ORDER BY count DESC, tags.name ASC`,
        )
        .all() as TagCount[];
    },
  };
}
```

```ts
import type Database from 'better-sqlite3';
import type { ClipboardItem, GetItemsQuery, ItemType } from '../../shared/types';
import { MAX_CONTENT_LENGTH } from '../../shared/config';
import type { TagsRepo } from './tags-repo';

export interface InsertItemInput {
  type: ItemType;
  content: string;
  sourceApp: string | null;
}

export interface ItemsRepo {
  insertItem(input: InsertItemInput): ClipboardItem;
  getItems(query: GetItemsQuery): ClipboardItem[];
  getById(id: number): ClipboardItem | null;
  toggleStar(id: number): void;
  deleteItem(id: number): void;
  pruneToLimit(limit: number): number;
}

interface ItemRow {
  id: number;
  type: ItemType;
  content: string;
  source_app: string | null;
  created_at: number;
  starred: number;
  pinned: number;
}

export function createItemsRepo(db: Database.Database, tagsRepo: TagsRepo): ItemsRepo {
  const insertStmt = db.prepare(
    `INSERT INTO items (type, content, source_app, created_at, starred, pinned)
     VALUES (?, ?, ?, ?, 0, 0)`,
  );
  const byIdStmt = db.prepare(`SELECT * FROM items WHERE id = ?`);
  const toggleStarStmt = db.prepare(`UPDATE items SET starred = 1 - starred WHERE id = ?`);
  const deleteStmt = db.prepare(`DELETE FROM items WHERE id = ?`);

  function rowToItem(row: ItemRow): ClipboardItem {
    return {
      id: row.id,
      type: row.type,
      content: row.content,
      sourceApp: row.source_app,
      createdAt: row.created_at,
      starred: row.starred === 1,
      pinned: row.pinned === 1,
      tags: tagsRepo.getTagsForItem(row.id),
    };
  }

  return {
    insertItem(input) {
      const content =
        input.content.length > MAX_CONTENT_LENGTH
          ? input.content.slice(0, MAX_CONTENT_LENGTH)
          : input.content;
      const createdAt = Date.now();
      const result = insertStmt.run(input.type, content, input.sourceApp, createdAt);
      return rowToItem(byIdStmt.get(result.lastInsertRowid) as ItemRow);
    },

    getItems(query) {
      const clauses: string[] = [];
      const params: unknown[] = [];
      let sql = 'SELECT DISTINCT items.* FROM items';

      if (query.tags && query.tags.length > 0) {
        sql +=
          ' JOIN item_tags ON item_tags.item_id = items.id JOIN tags ON tags.id = item_tags.tag_id';
        clauses.push(`tags.name IN (${query.tags.map(() => '?').join(',')})`);
        params.push(...query.tags.map((t) => t.trim().toLowerCase()));
      }
      if (query.onlyStarred) clauses.push('items.starred = 1');
      if (query.type) {
        clauses.push('items.type = ?');
        params.push(query.type);
      }
      if (query.search) {
        clauses.push('items.content LIKE ?');
        params.push(`%${query.search}%`);
      }
      if (clauses.length > 0) sql += ' WHERE ' + clauses.join(' AND ');
      sql += ' ORDER BY items.pinned DESC, items.created_at DESC';

      const rows = db.prepare(sql).all(...params) as ItemRow[];
      return rows.map(rowToItem);
    },

    getById(id) {
      const row = byIdStmt.get(id) as ItemRow | undefined;
      return row ? rowToItem(row) : null;
    },

    toggleStar(id) {
      toggleStarStmt.run(id);
    },

    deleteItem(id) {
      deleteStmt.run(id);
    },

    pruneToLimit(limit) {
      const result = db
        .prepare(
          `DELETE FROM items WHERE id IN (
             SELECT id FROM items WHERE starred = 0
             ORDER BY created_at DESC LIMIT -1 OFFSET ?
           )`,
        )
        .run(limit);
      return result.changes;
    },
  };
}
```

```ts
export function parseFileDropBuffer(buf: Buffer): string[] {
  if (!buf || buf.length === 0) return [];
  try {
    const text = buf.toString('ucs2');
    return text
      .split('\u0000')
      .map((s) => s.trim())
      .filter(Boolean);
  } catch {
    return [];
  }
}
```

```ts
import { hashContent } from './hash';
import { parseFileDropBuffer } from './file-drop';
import type { ClipboardReader, ClipboardSnapshot } from './types';
import { POLL_INTERVAL_MS } from '../../shared/config';

export type CaptureCallback = (snapshot: ClipboardSnapshot) => void;

export class ClipboardWatcher {
  private lastHash: string | null = null;
  private intervalId: ReturnType<typeof setInterval> | null = null;

  constructor(
    private reader: ClipboardReader,
    private onCapture: CaptureCallback,
    private pollMs: number = POLL_INTERVAL_MS,
  ) {}

  pollOnce(): void {
    const snapshot = this.capture();
    if (!snapshot || snapshot.hash === this.lastHash) return;
    this.lastHash = snapshot.hash;
    this.onCapture(snapshot);
  }

  private capture(): ClipboardSnapshot | null {
    if (this.reader.hasFileDrop()) {
      const paths = parseFileDropBuffer(this.reader.readFileDropBuffer());
      if (paths.length === 0) return null;
      return { type: 'file', paths, hash: hashContent(paths.join('|')) };
    }
    if (this.reader.hasImage()) {
      const pngBuffer = this.reader.readImagePNG();
      if (!pngBuffer || pngBuffer.length === 0) return null;
      return { type: 'image', pngBuffer, hash: hashContent(pngBuffer) };
    }
    const text = this.reader.readText();
    if (!text || text.trim().length === 0) return null;
    return { type: 'text', content: text, hash: hashContent(text) };
  }

  start(): void {
    if (this.intervalId) return;
    this.intervalId = setInterval(() => this.pollOnce(), this.pollMs);
  }

  stop(): void {
    if (this.intervalId) {
      clearInterval(this.intervalId);
      this.intervalId = null;
    }
  }
}
```

```ts
import fs from 'node:fs';
import path from 'node:path';
import { hashContent } from './hash';

export function saveImagePreview(previewsDir: string, pngBuffer: Buffer): string {
  fs.mkdirSync(previewsDir, { recursive: true });
  const filePath = path.join(previewsDir, `${hashContent(pngBuffer)}.png`);
  if (!fs.existsSync(filePath)) {
    fs.writeFileSync(filePath, pngBuffer);
  }
  return filePath;
}
```

```ts
import type { ClipboardItem } from '../../shared/types';
import type { ItemsRepo } from '../storage/items-repo';
import type { ClipboardSnapshot } from './types';
import { saveImagePreview } from './image-store';

export interface CapturePipelineDeps {
  itemsRepo: ItemsRepo;
  previewsDir: string;
  retentionLimit: number;
  onItemAdded: (item: ClipboardItem) => void;
}

export function createCapturePipeline(deps: CapturePipelineDeps) {
  return function handleSnapshot(snapshot: ClipboardSnapshot): void {
    let content: string;
    if (snapshot.type === 'image') {
      content = saveImagePreview(deps.previewsDir, snapshot.pngBuffer);
    } else if (snapshot.type === 'file') {
      content = JSON.stringify(snapshot.paths);
    } else {
      content = snapshot.content;
    }

    const item = deps.itemsRepo.insertItem({
      type: snapshot.type,
      content,
      sourceApp: null,
    });
    deps.itemsRepo.pruneToLimit(deps.retentionLimit);
    deps.onItemAdded(item);
  };
}
```

```ts
import { create } from 'zustand';
import type { ClipboardItem, ItemType } from '../../shared/types';

export type ActiveTab = 'history' | 'starred';

interface ClipboardState {
  items: ClipboardItem[];
  search: string;
  typeFilter: ItemType | null;
  tagFilters: string[];
  activeTab: ActiveTab;
  fetchItems: () => Promise<void>;
  setSearch: (value: string) => void;
  setTypeFilter: (type: ItemType | null) => void;
  toggleTagFilter: (tag: string) => void;
  setActiveTab: (tab: ActiveTab) => void;
  copyItem: (id: number) => Promise<void>;
  pasteItem: (id: number) => Promise<void>;
  toggleStar: (id: number) => Promise<void>;
  addTag: (id: number, tag: string) => Promise<void>;
  removeTag: (id: number, tag: string) => Promise<void>;
  deleteItem: (id: number) => Promise<void>;
  subscribeToNewItems: () => () => void;
}

export const useClipboardStore = create<ClipboardState>((set, get) => ({
  items: [],
  search: '',
  typeFilter: null,
  tagFilters: [],
  activeTab: 'history',

  async fetchItems() {
    const { search, typeFilter, tagFilters, activeTab } = get();
    const items = await window.copycat.getItems({
      search: search || undefined,
      type: typeFilter ?? undefined,
      tags: tagFilters.length > 0 ? tagFilters : undefined,
      onlyStarred: activeTab === 'starred' || undefined,
    });
    set({ items });
  },

  setSearch(value) {
    set({ search: value });
    void get().fetchItems();
  },

  setTypeFilter(type) {
    set({ typeFilter: type });
    void get().fetchItems();
  },

  toggleTagFilter(tag) {
    const current = get().tagFilters;
    const next = current.includes(tag)
      ? current.filter((t) => t !== tag)
      : [...current, tag];
    set({ tagFilters: next });
    void get().fetchItems();
  },

  setActiveTab(tab) {
    set({ activeTab: tab });
    void get().fetchItems();
  },

  async copyItem(id) {
    await window.copycat.copyItem(id);
  },

  async pasteItem(id) {
    await window.copycat.pasteItem(id);
  },

  async toggleStar(id) {
    await window.copycat.toggleStar(id);
    await get().fetchItems();
  },

  async addTag(id, tag) {
    await window.copycat.addTag(id, tag);
    await get().fetchItems();
  },

  async removeTag(id, tag) {
    await window.copycat.removeTag(id, tag);
    await get().fetchItems();
  },

  async deleteItem(id) {
    await window.copycat.deleteItem(id);
    await get().fetchItems();
  },

  subscribeToNewItems() {
    return window.copycat.onItemAdded(() => {
      void get().fetchItems();
    });
  },
}));
```

```tsx
export function TagEditor({ item, onAddTag, onRemoveTag }: TagEditorProps) {
  const [draft, setDraft] = useState('');

  function submit() {
    const tag = draft.trim();
    if (tag) onAddTag(item.id, tag);
    setDraft('');
  }

  return (
    <div className="tag-editor">
      {item.tags.map((tag) => (
        <span key={tag} className="tag-chip">
          {tag}
          <button aria-label={`remove-${tag}`} onClick={() => onRemoveTag(item.id, tag)}>
            ×
          </button>
        </span>
      ))}
      <input
        value={draft}
        onChange={(e) => setDraft(e.target.value)}
        onKeyDown={(e) => {
          if (e.key === 'Enter') submit();
        }}
        placeholder="Add tag..."
      />
    </div>
  );
}
```

```tsx
import { useEffect } from 'react';
import { useClipboardStore } from './store/useClipboardStore';
import { SearchBar } from './components/SearchBar';

import { TagFilterRow } from './components/TagFilterRow';

    items,
    search,
    activeTab,
    tagFilters,
    setSearch,
    setActiveTab,
    toggleTagFilter,
    fetchItems,
    copyItem,
    toggleStar,
    deleteItem,
    subscribeToNewItems,
  } = useClipboardStore();

  useEffect(() => {
    void fetchItems();
    return subscribeToNewItems();
  }, []);

  const tagCounts = Array.from(
    items.flatMap((item) => item.tags).reduce((acc, tag) => {
      acc.set(tag, (acc.get(tag) ?? 0) + 1);
      return acc;
    }, new Map<string, number>()),
  ).map(([name, count]) => ({ name, count }));

  return (

      <div className="toolbar">
        <SearchBar value={search} onChange={setSearch} />
        <TabBar active={activeTab} onChange={setActiveTab} />
      </div>
      <TagFilterRow tags={tagCounts} activeTags={tagFilters} onToggle={toggleTagFilter} />

        items={items}

        onToggleStar={toggleStar}
```
